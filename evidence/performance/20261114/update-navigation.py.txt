def update(name, transform):
    path = ROOT / name
    raw = path.read_bytes()
    newline = "\r\n" if b"\r\n" in raw else "\n"
    text = raw.decode("utf-8").replace("\r\n", "\n")
    path.write_bytes(transform(text).replace("\n", newline).encode("utf-8"))


def paragraph(text, prefix, replacement):
    parts = text.split("\n\n")
    matches = [index for index, value in enumerate(parts) if value.startswith(prefix)]
    assert len(matches) == 1, prefix
    parts[matches[0]] = replacement
    return "\n\n".join(parts)
