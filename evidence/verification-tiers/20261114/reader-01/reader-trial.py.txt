import argparse
import hashlib
import json
import subprocess
import sys

from datetime import UTC, datetime
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
sys.path.insert(0, str(ROOT))
from evaluations.host_trials import codex_command
from evaluations.request_review_trials import run as native_run

DOCUMENTS = ("README.md", "evidence/README.md", "docs/VERIFICATION.md", "docs/EVIDENCE-INTEGRITY.md")
SOURCES = ("evaluations/host_trials.py", "evaluations/request_review_trials.py", "evaluations/native_trial_records.py")
QUESTIONS = [
    {"id": "inspect", "question": "Find the current verification guide from the public README. Execute its appropriate read-only command to inspect existing evidence without rebuilding it; retain exact stdout/stderr and actual child exit code under work/. Explain what the result means, including current versus historical applicability. Return inspection metadata as specified below."},
    {"id": "focused", "question": "A developer changes consumer-index metadata behavior. Which exact documented focused command is useful, what does it write, and what can a passing result establish? Explain why running all unittest discovery is a different side-effect choice. Do not execute either test route in this reader assignment."},
    {"id": "rebuild", "question": "Explain the full verifier's order, changed outputs, archive coverage and failure behavior. Is appending --help a safe way to inspect its options? What important effects and locations remain outside its archive? Do not run rebuilders or their --help variants."},
    {"id": "tiers", "question": "Distinguish local implementation/distribution checks, native run completion, grading retained work, useful asset/consumer outcomes and human acceptance. Which native preparation/run routes may contact a model or the official provider, and what does an old grade or current integrity result fail to establish? Do not run a native or provider assignment."},
]


def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()


def write(path, value):
    path = Path(path)
    path.parent.mkdir(parents=True, exist_ok=True)
    with path.open("x", encoding="utf-8", newline="\n") as stream:
        json.dump(value, stream, ensure_ascii=False, indent=2)
        stream.write("\n")


def inventory(names):
    result = {}
    for name in names:
        path = ROOT / name
        if not path.is_file():
            result[name] = {"missing": True}
        else:
            stat = path.stat()
            result[name] = {"sha256": sha(path), "bytes": stat.st_size, "mtime_ns": stat.st_mtime_ns}
    return result


def tree_hash(value):
    return hashlib.sha256(json.dumps(value, sort_keys=True, separators=(",", ":")).encode()).hexdigest()


def prepare(attempt, selected=None, documentation_only=False):
    attempt.mkdir(parents=True, exist_ok=False)
    consumer = Path(tempfile.mkdtemp(prefix="pixl evidence reader ")).resolve()
    (consumer / "work").mkdir()
    guidance = f"""You are an unfamiliar evidence reader in a bounded documentation usability evaluation.
The repository at {ROOT.as_posix()} is read-only source material, not your workspace.
Begin with its README.md. Discover the current verification guide from that page, then
follow local document links to answer the questions in ASSIGNMENT.md. You have no
prior project context or supplied expected answers. Do not recursively search or
enumerate dated evidence directories, use filename/date guessing, search outside
this repository, browse the web, delegate, inspect credentials or persistent memory,
edit source/evidence, rebuild a kit, run tests, call a provider or create art.
You may search within a document you opened and read linked receipts. The documented
read-only integrity command is required for the inspection question; save exact
stdout/stderr and actual subprocess exit status only in this
scratch workspace's work/ directory. Source paths in your answer are repository-relative.
Write only under {consumer.as_posix()}/work plus the final NAVIGATION-RESULT.json here.
Human acceptance and product acceptance are separate from this reader exercise.
"""
    questions = QUESTIONS if not selected else [row for row in QUESTIONS if row["id"] in selected]
    if selected and set(selected) != {row["id"] for row in questions}:
        raise ValueError("Unknown requested question")
    assignment = "# Choose verification from the public entrypoint\n\n" + "\n\n".join(
        f"{n}. **{question['id']}** — {question['question']}" for n, question in enumerate(questions, 1))
    assignment += """\n\nReturn NAVIGATION-RESULT.json with version:1, status:'complete' or 'incomplete',
entrypoint, a route array naming the repository-relative documents followed, and
inspection:{command:[strings], exit_code:integer, stdout_path:'work/relative-file', stderr_path:'work/relative-file'}. These paths are relative to your scratch root. Include
answers (one per question). Each answer has id, conclusion, evidence (an array of
{path,sha256} references to exact local files), limits (strings), and prior_failures
(exact file references where applicable). Quote only small relevant facts; do not
copy large receipts. Cite exact receipt paths and hashes rather than relying only
on an index summary. If a fact cannot be found, state that gap; do not infer a pass.
At the end, report any confusing navigation, contradictory current-facing wording,
or missing route in navigation_observations (strings). No post-launch help will be
provided. Do not write the evaluator's grade or alter any original artifact.
"""
    (consumer / "AGENTS.md").write_text(guidance, encoding="utf-8")
    (consumer / "ASSIGNMENT.md").write_text(assignment, encoding="utf-8")
    (attempt / "prompt.txt").write_text(guidance + "\nRead ASSIGNMENT.md and complete this evidence-reading task.\n", encoding="utf-8")
    write(attempt / "questions.json", questions)
    write(attempt / "locations.json", {"consumer": str(consumer), "root": str(ROOT)})
    names = sorted(set(subprocess.check_output(["git", "ls-files", "-z"], cwd=ROOT).decode().split("\0")) - {""} | set(DOCUMENTS))
    if documentation_only:
        names = sorted({name for name in names if Path(name).suffix.lower() == ".md"} | set(DOCUMENTS) | set(SOURCES) | {"evaluations/evidence-claims.json", "plugins/pixl/bundle-manifest.json"})
    protected = inventory(names)
    with zipfile.ZipFile(attempt / "protected-inputs.zip", "x", zipfile.ZIP_DEFLATED) as archive:
        archive.writestr("files.json", json.dumps(protected, separators=(",", ":")))
    for name in DOCUMENTS + SOURCES:
        target = attempt / "source-snapshot" / name
        target.parent.mkdir(parents=True, exist_ok=True)
        target.write_bytes((ROOT / name).read_bytes())
    (attempt / "reader-trial.py.txt").write_bytes(Path(__file__).read_bytes())
    native = codex_command()
    version = subprocess.check_output(native + ["--version"], text=True).strip()
    command = native + ["exec", "--ignore-user-config", "--ignore-rules", "--json",
                        "--skip-git-repo-check", "--approve-for-me", "-C", str(consumer),
                        "-c", "features.apps=false", "-c", "features.plugins=false",
                        "-c", 'web_search="disabled"', "-"]
    write(attempt / "launch.json", {"host": "codex", "case": "qol09-verification-guide-reader", "cwd": str(consumer),
                                    "command": command, "host_version": version})
    write(attempt / "identity.json", {"source_revision": subprocess.check_output(["git", "rev-parse", "HEAD"], cwd=ROOT, text=True).strip(),
        "observed_at": datetime.now(UTC).isoformat(), "documents": {name: sha(ROOT / name) for name in DOCUMENTS},
        "sources": {name: sha(ROOT / name) for name in SOURCES}, "runner_sha256": sha(Path(__file__)),
        "prompt_sha256": sha(attempt / "prompt.txt"), "questions_sha256": sha(attempt / "questions.json"),
        "protected_inputs": {"path": "protected-inputs.zip", "sha256": sha(attempt / "protected-inputs.zip"),
                             "selection": "tracked Markdown, navigation pages, runner sources, registry and bundle manifest" if documentation_only else "all tracked files plus the new navigation index",
                             "count": len(protected), "bytes": sum(row.get("bytes", 0) for row in protected.values()), "tree_sha256": tree_hash(protected)},
        "scratch_inputs": {name: sha(consumer / name) for name in ("AGENTS.md", "ASSIGNMENT.md")},
        "scope": "Unfamiliar native reader follows the public verification guide and performs the documented read-only check; no inherited task answers. Public traces and input preservation support the task boundary, not an OS sandbox."})
    print(json.dumps({"attempt": str(attempt), "host_version": version, "protected_files": len(protected)}))


def retain(attempt):
    identity = json.loads((attempt / "identity.json").read_bytes())
    consumer = Path(json.loads((attempt / "locations.json").read_bytes())["consumer"])
    destination = attempt / "project"
    if destination.exists():
        raise ValueError("Use a fresh retained output, never overwrite a previous attempt")
    for path in consumer.rglob("*"):
        if path.is_file():
            target = destination / path.relative_to(consumer)
            target.parent.mkdir(parents=True, exist_ok=True)
            target.write_bytes(path.read_bytes())
    with zipfile.ZipFile(attempt / "protected-inputs.zip") as archive:
        before = json.loads(archive.read("files.json"))
    after = inventory(before)
    changed = [name for name, row in before.items() if after[name] != row]
    write(attempt / "preservation.json", {"status": "pass" if not changed else "fail", "files": len(before),
          "before_tree_sha256": tree_hash(before), "after_tree_sha256": tree_hash(after), "changed": changed,
          "scratch_inputs_unchanged": all(sha(consumer / name) == digest for name, digest in identity["scratch_inputs"].items()),
          "scope": identity["protected_inputs"].get("selection", "all tracked files plus the new navigation index") + "; exact bytes and mtimes. New evaluation outputs are outside this preserved set."})
    print(json.dumps({"preservation": "pass" if not changed else "fail", "changed_count": len(changed), "result": (destination / "NAVIGATION-RESULT.json").is_file()}))


if __name__ == "__main__":
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("action", choices=("prepare", "run", "retain"))
    parser.add_argument("--attempt", type=Path, required=True)
    parser.add_argument("--question", action="append")
    parser.add_argument("--documentation-only", action="store_true")
    args = parser.parse_args()
    attempt = args.attempt.resolve()
    if args.action == "prepare":
        prepare(attempt, args.question, args.documentation_only)
    elif args.action == "run":
        native_run(attempt, timeout=900)
    else:
        retain(attempt)
