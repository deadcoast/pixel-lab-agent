import hashlib,json,os,runpy,sys
from copy import deepcopy
from pathlib import Path
import pixl
from pixl import codec,diagnostics,graph,store

mode,project_text,fixture_text = sys.argv[1:4]
project = Path(project_text).resolve()
folder = project / ".pixl"
modules = {name:str(Path(module.__file__).resolve()) for name,module in
           (("pixl",pixl),("store",store),("codec",codec),("diagnostics",diagnostics),("graph",graph))}

def document_info(doc):
    return {"hash":codec.digest(doc),"revision":doc.revision,"task_state":doc.records["task"].fields["st"].value}

def marker(phase):
    path = Path(sys.argv[4])
    value = {"pid":os.getpid(),"phase":phase,"modules":modules,"state":document_info(store.load(folder))}
    with path.open("x",encoding="utf-8") as stream:
        json.dump(value,stream,indent=2)
    print(json.dumps(value),flush=True)

if mode == "setup":
    sample = runpy.run_path(fixture_text)["sample"]
    initial = sample(project)
    store.initialize(folder,initial,root=project)
    candidate = deepcopy(initial)
    candidate.revision += 1
    candidate.records["task"].fields["st"] = codec.Symbol("active")
    with (project/"candidate-1.px").open("xb") as stream:
        stream.write(codec.encode(candidate))
    print(json.dumps({"initial":document_info(initial),"candidate":document_info(candidate),"modules":modules}))
elif mode == "hold":
    with store.lock(folder):
        marker("lock_held_before_publication")
        sys.stdin.buffer.read(1)
elif mode == "publish":
    current = store.load(folder)
    candidate = codec.parse((project/"candidate-1.px").read_bytes())
    original_atomic = store._atomic
    def stop_after_head(path,data):
        original_atomic(path,data)
        if Path(path).resolve() == (folder/"state.px").resolve():
            marker("head_replaced_before_lock_cleanup")
            sys.stdin.buffer.read(1)
    store._atomic = stop_after_head
    store.commit(folder,candidate,codec.digest(current),root=project)
elif mode == "next":
    current = store.load(folder)
    candidate = deepcopy(current)
    candidate.revision += 1
    candidate.records["task"].fields["st"] = codec.Symbol("blocked")
    with (project/"candidate-2.px").open("xb") as stream:
        stream.write(codec.encode(candidate))
    print(json.dumps({"current":document_info(current),"candidate":document_info(candidate),"modules":modules}))
else:
    raise ValueError("Unknown controlled worker mode")
