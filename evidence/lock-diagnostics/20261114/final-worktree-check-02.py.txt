"""Read-only whitespace and credential-shape scan for this increment's files."""
import argparse
from datetime import UTC, datetime
import hashlib
import io
import json
from pathlib import Path
import re
import subprocess
import tarfile
import zipfile

ROOT = Path(__file__).resolve().parents[3]
PATTERNS = (re.compile(rb"pixl_[0-9a-fA-F]{32}\b"),
            re.compile(rb"-----BEGIN (?:RSA |EC |OPENSSH )?PRIVATE KEY-----"))


def main():
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--out", required=True, type=Path)
    args = parser.parse_args()
    if args.out.exists():
        raise ValueError("Preserve earlier checks; select a fresh output")
    tracked = subprocess.check_output(["git", "diff", "--name-only", "HEAD"], cwd=ROOT, text=True).splitlines()
    untracked = subprocess.check_output(["git", "ls-files", "--others", "--exclude-standard"], cwd=ROOT, text=True).splitlines()
    names = sorted(set(tracked + untracked))
    findings, count, archive_members = [], 0, 0
    def scan(name, data):
        nonlocal archive_members
        if any(pattern.search(data) for pattern in PATTERNS):
            findings.append(name)
        if name.endswith((".zip", ".whl")):
            with zipfile.ZipFile(io.BytesIO(data)) as archive:
                for member in archive.infolist():
                    if not member.is_dir():
                        archive_members += 1
                        scan(name + "::" + member.filename, archive.read(member))
        elif name.endswith(".tar.gz"):
            with tarfile.open(fileobj=io.BytesIO(data), mode="r:gz") as archive:
                for member in archive:
                    if member.isfile():
                        archive_members += 1
                        scan(name + "::" + member.name, archive.extractfile(member).read())
    for name in names:
        path = ROOT / name
        if path.is_file():
            count += 1
            scan(name, path.read_bytes())
    retained = [
        "evidence/lock-diagnostics/20261114/implementation-01/stderr.txt",
        "evidence/lock-diagnostics/20261114/process-03/copied kit/resources/docs/research/PIXEL-ART-ASSETS.md",
        "evidence/lock-diagnostics/20261114/process-03/copied kit/resources/training/original-docs/PIXEL-ART-ASSETS.md",
        "evidence/lock-diagnostics/20261114/process-03/copied kit/skills/pixl/references/artistic-authority.md",
        "evidence/lock-diagnostics/20261114/process-03/copied kit/skills/pixl/references/state-authority.md",
    ]
    prior = json.loads((ROOT / "evidence/lock-diagnostics/20261114/integrity-02/before.json").read_bytes())
    exceptions = []
    for name in retained:
        digest = hashlib.sha256((ROOT / name).read_bytes()).hexdigest()
        intact = digest == prior[name]["sha256"]
        canonical = None
        if "/copied kit/" in name:
            canonical = "plugins/pixl/" + name.split("/copied kit/", 1)[1]
            intact = intact and (ROOT / name).read_bytes() == (ROOT / canonical).read_bytes()
        exceptions.append({"path": name, "sha256": digest, "unchanged_since_integrity": intact,
                           "canonical_copy": canonical,
                           "reason": "Preserve raw failed-test output or exact tested kit bytes; do not normalize retained evidence."})
    whitespace = subprocess.run(["git", "-c",
        "core.whitespace=blank-at-eol,blank-at-eof,space-before-tab,cr-at-eol",
        "diff", "--check", "HEAD", "--", ".", *[":(exclude)" + name for name in retained]], cwd=ROOT, capture_output=True, text=True)
    result = {"version": 1, "observed_at": datetime.now(UTC).isoformat(),
        "scope": "Changed/untracked files and new ZIP/wheel/sdist members including nested archives only; credential shapes, not a complete secret audit. Existing CRLF conventions are preserved.",
        "script_sha256": hashlib.sha256(Path(__file__).read_bytes()).hexdigest(),
        "files_scanned": count, "archive_members_scanned": archive_members,
        "credential_shape_paths": findings, "whitespace_exit_code": whitespace.returncode,
        "whitespace_output": whitespace.stdout + whitespace.stderr,
        "retained_raw_whitespace": exceptions,
        "predecessor": "evidence/lock-diagnostics/20261114/final-worktree-check-01.json",
        "whitespace_scope": "All changed paths except five exact preserved raw-output/copied-kit files, whose bytes are verified against the integrity snapshot and canonical kit where applicable. The original broad whitespace failure remains unchanged.",
        "status": "pass" if not findings and whitespace.returncode == 0 and all(item["unchanged_since_integrity"] for item in exceptions) else "fail"}
    with args.out.open("x", encoding="utf-8") as stream:
        json.dump(result, stream, indent=2)
        stream.write("\n")
    print(json.dumps({key: result[key] for key in ("status", "files_scanned", "archive_members_scanned", "credential_shape_paths", "whitespace_exit_code")}))
    raise SystemExit(0 if result["status"] == "pass" else 1)


if __name__ == "__main__":
    main()
