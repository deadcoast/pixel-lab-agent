import json
from pathlib import Path
from hashlib import sha256
from PIL import Image

ROOT=Path(__file__).resolve().parents[1]/'evidence/pixellab-family-production/20261108/project'
WORK=ROOT/'work/live-family'
frames=[Image.open(WORK/f'working-frame-{i}.png').convert('RGBA') for i in range(5)]
assert all(f.size==(16,16) for f in frames)
sheet=Image.new('RGBA',(64,16))
for index, frame in enumerate(frames[1:]):
    sheet.paste(frame,(index*16,0))
path=WORK/'working-animation-01-raw.png'
assert not path.exists()
sheet.save(path)
record={'operation':'lossless frame selection and horizontal assembly; no drawing, scaling or color edits','provider_job':'7fb6c786-322c-4ebd-847e-74f978ddfffa','source_indices':[1,2,3,4],'retained_input_index':0,'duration_ms':[100]*4,'raw_frames':[{'index':i,'path':f'work/live-family/working-frame-{i}.png','sha256':sha256((WORK/f'working-frame-{i}.png').read_bytes()).hexdigest()} for i in range(5)],'output':path.name,'sha256':sha256(path.read_bytes()).hexdigest()}
(WORK/'working-animation-01-assembly.json').write_text(json.dumps(record,indent=2)+'\n',encoding='utf-8')
print(json.dumps(record))
