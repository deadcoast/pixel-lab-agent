"""Independently verify native current-kit export and comparison continuity."""
import json
import sys
from hashlib import sha256
from pathlib import Path

repo = Path(__file__).resolve().parents[1]
attempt = repo/'evidence/criteria-handoff/20261109/native/claude-export-01'
project = attempt/'project'
kit = project/'.agents/skills/pixl-kit'
sys.path.insert(0,str(kit/'runtime'))
from pixl import production_review

def read(path):return json.loads(path.read_text(encoding='utf-8'))
def digest(path):return sha256(path.read_bytes()).hexdigest()
checks=[]
def check(name,passed,detail=None):checks.append({'check':name,'pass':bool(passed),'detail':detail})
identity=read(attempt/'identity.json');record=read(attempt/'record.json')
changed=[p for p,h in identity['inputs'].items() if not (project/p).is_file() or digest(project/p)!=h]
check('All pre-run consumer files unchanged',not changed,{'count':len(identity['inputs']),'changed':changed})
changed=[p for p,h in identity['package_files'].items() if not (kit/p).is_file() or digest(kit/p)!=h]
check('Frozen final kit unchanged',not changed,changed)
prior=Path(identity['predecessor']);prior_record=read(prior/'record.json')
changed=[p for p,h in prior_record['artifacts'].items() if not (prior/'project'/p).is_file() or digest(prior/'project'/p)!=h]
check('Completed delegated review remains unchanged',not changed,changed)
check('Native caller completed successfully',record['returncode']==0,record['execution_status'])
work=project/'work/export-review-01'
raw=read(work/'request-review-raw.txt');formatted=read(work/'request-review.json')
old_raw=read(project/'work/criteria-review-01/request-review-raw.txt')
old_formatted=read(project/'work/criteria-review-01/request-review.json')
check('Fresh formatted report equals exact raw UTF-8 JSON',raw==formatted)
check('Prior raw and formatted native reports were already identical',old_raw==old_formatted)
current=production_review.report(project,'.pixl/assets/queue.ready/state.px','authoring-01.plan','authoring-01.working.join-01')
check('Saved fresh report equals actual final-runtime report',current==raw)
def semantic(value):
    return {**{k:v for k,v in value.items() if k not in ('review_scope_sha256','sources')},
            'sources':[{k:v for k,v in s.items() if k!='absolute_path'} for s in value['sources']]}
check('Prior-to-current report changes only fingerprint and observed absolute locations',semantic(old_raw)==semantic(raw))
old_caller=read(project/'work/criteria-review-01/caller-comparison.json')
caller=read(work/'caller-comparison.json')
check('All 20 caller criterion comparisons preserved verbatim',len(caller['criterion_comparisons'])==20 and caller['criterion_comparisons']==old_caller['criterion_comparisons'])
check('Discrepancy resolutions, verdict and reduced-motion scope preserved',all(caller[k]==old_caller[k] for k in ('discrepancies_with_specialist','verdict','reduced_motion_disposition')))
coverage=production_review.report(project,'.pixl/assets/queue.ready/state.px','authoring-01.plan','authoring-01.working.join-01','work/export-review-01/caller-comparison.json')['comparison_coverage']
check('Final-runtime coverage complete for exact current identities',coverage['complete'],coverage)
check('Saved coverage matches fresh evaluation',read(work/'request-review-coverage.json')['comparison_coverage']==coverage)
check('Current changed requirement and blocked prepared job retained',raw['diagnostics']==old_raw['diagnostics'] and raw['request']['state']=='prepared' and raw['request']['remote_id'] is None and raw['request']['next_action']=='resolve_context')
check('Native decoder observation corrects erroneous assignment premise',read(work/'decoder-observation.json')['prior_formatted_equals_prior_raw_parsed'])
result=read(project/'EXPORT-RESULT.json')
check('Explicit caller return retained',bool(result.get('status')))
events=[json.loads(l) for l in (attempt/'stdout.jsonl').read_text(encoding='utf-8').splitlines() if l.strip()]
calls=[c for e in events for c in e.get('message',{}).get('content',[]) if isinstance(c,dict) and c.get('type')=='tool_use' and c.get('name')=='mcp__pixl-local__pixl_request_review']
check('Native caller used report and coverage MCP operations',len(calls)==2 and any('comparison' in c['input'] for c in calls))
value={'scope':'Current-kit byte/identity continuity after completed native delegated semantic review; no repeated subjective judgment or production.',
       'verdict':'pass' if all(c['pass'] for c in checks) else 'fail','checks':checks,
       'passed':sum(c['pass'] for c in checks),'total':len(checks),'elapsed_seconds':record['elapsed_seconds'],
       'result':result,'runtime':str(production_review.__file__),'review_scope_sha256':raw['review_scope_sha256']}
out=repo/'evidence/criteria-handoff/independent-review/export-checks.json'
out.write_text(json.dumps(value,indent=2)+'\n',encoding='utf-8')
print(json.dumps({k:value[k] for k in ('verdict','passed','total','elapsed_seconds')}))
print(json.dumps({'failures':[c for c in checks if not c['pass']]}))
