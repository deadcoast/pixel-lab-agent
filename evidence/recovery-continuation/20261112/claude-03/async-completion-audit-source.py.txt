def sha(data):
    return hashlib.sha256(data).hexdigest()


def reference(path):
    data = path.read_bytes()
    return {'path': path.relative_to(ROOT).as_posix(), 'sha256': sha(data), 'bytes': len(data)}


def tag(text, name):
    matches = re.findall(r'<' + name + r'>(.*?)</' + name + r'>', text, re.DOTALL)
    if len(matches) != 1:
        raise ValueError('Expected one host notification tag: ' + name)
    return matches[0]
