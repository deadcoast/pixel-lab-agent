def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path, needle=None, before=0, after=0):
    result = {"path": path.relative_to(ROOT).as_posix(), "sha256": sha(path)}
    if needle:
        lines = path.read_text(encoding="utf-8").splitlines()
        index = next(i for i, line in enumerate(lines) if needle in line)
        first, last = max(0, index - before), min(len(lines), index + after + 1)
        result.update({"lines": [first + 1, last], "quote": "\n".join(lines[first:last])})
    return result


def event(name, line):
    path = NATIVE / "native-transcripts" / name
    row = json.loads(path.read_text(encoding="utf-8").splitlines()[line - 1])
    result = ref(path)
    result.update({"line": line, "event_type": row.get("type"), "message_role": row.get("message", {}).get("role")})
    return result
