"""Retain the same unfinished native assignment with the user's selected model."""
import json
import os
import subprocess
import sys
from datetime import datetime, timezone
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
BASE = Path(__file__).parent
sys.path.insert(0, str(ROOT))

from evaluations import skill_handoff_trials as handoff
from evaluations.host_trials import inventory, write_json
from evaluations.native_trial_records import prepare_resume, reference


def main():
    old_base = BASE.parent / "literal-reconciliation-20261116"
    previous = old_base / "native-02-resume-01"
    destination = BASE / "native-01"
    old_protocol = old_base / "protocol-continuation-02.json"
    protocol = json.loads(old_protocol.read_bytes())
    old_identity = json.loads((previous / "identity.json").read_bytes())
    old_launch = json.loads((previous / "launch.json").read_bytes())
    consumer = Path(old_launch["cwd"]).resolve()
    current = handoff._files(consumer, exclude_kit=True)
    assert len(current) == 167
    assert current == handoff._files(previous / "project")
    assert current == json.loads((previous / "preservation.json").read_bytes())["outputs"]
    assert handoff._files(consumer / handoff.KIT) == old_identity["package_files"]
    assert len(old_identity["package_files"]) == 261
    predecessor_files = inventory(previous, ("__pycache__",))

    config_dir = Path(os.environ.get("CLAUDE_CONFIG_DIR", str(Path.home() / ".claude")))
    user_settings_path = config_dir / "settings.json"
    settings_before = reference(user_settings_path)
    user_settings = json.loads(user_settings_path.read_bytes())
    model = user_settings["model"]
    assert model == "claude-opus-5-5", "Reassess changed user model selection before inference"
    effort = user_settings["effortLevel"]
    assert effort == "xhigh"
    prepare_resume(previous, destination, protocol["assignment"]["resume_prompt"],
                   "a94b4dc767b65b243", model=model)
    launch_path = destination / "launch.json"
    launch = json.loads(launch_path.read_bytes())
    scoped_settings = destination / "model-settings.json"
    write_json(scoped_settings, {
        "effortLevel": effort,
        "env": {"CLAUDE_CODE_SUBAGENT_MODEL": model, "CLAUDE_CODE_SUBAGENT_MODEL_FORCE": "1"},
    })
    launch["command"] += ["--settings", str(scoped_settings.resolve())]
    launch["configuration_files"]["invocation_model_settings"] = str(scoped_settings.resolve())
    write_json(launch_path, launch)

    auth = subprocess.run([launch["command"][0], "--setting-sources", "", "--settings",
                           str(scoped_settings.resolve()), "auth", "status", "--json"],
                          cwd=consumer, capture_output=True, text=True, check=True)
    auth_status = json.loads(auth.stdout)
    selected_auth = {key: auth_status.get(key) for key in
                     ("loggedIn", "authMethod", "apiProvider", "subscriptionType")}
    assert selected_auth == {"loggedIn": True, "authMethod": "claude.ai",
                             "apiProvider": "firstParty", "subscriptionType": "max"}
    write_json(BASE / "host-selection.json", {
        "observed_at_utc": datetime.now(timezone.utc).isoformat(),
        "auth": selected_auth, "configured_model": model, "configured_effort": effort,
        "user_settings_identity": settings_before,
        "user_settings_unchanged": reference(user_settings_path) == settings_before,
        "scope": "Native CLI auth status only; no inference, credentials, account identifiers or new sign-in.",
        "cause": "Isolated evaluation excluded user settings and omitted --model; retained replies used Fable. The model-credit failure did not establish exhausted Max included access.",
        "documentation": ["https://code.claude.com/docs/en/model-config", "https://code.claude.com/docs/en/sub-agents"],
    })

    protocol_path = BASE / "protocol.json"
    write_json(protocol_path, {
        "schema_version": 1, "status": "frozen_before_inference",
        "frozen_at_utc": datetime.now(timezone.utc).isoformat(),
        "inherits_assignment_and_all_criteria": reference(old_protocol),
        "assignment": protocol["assignment"],
        "caller": old_identity["session_id"], "child": old_identity["child_id"],
        "predecessor": reference(previous / "continuation.json"),
        "change": "Use current user-selected Opus model and effort explicitly despite isolated settings; pin subagents to that model for this invocation. Preserve installed Max OAuth, original native parent/child, consumer, frozen kit, prior failures and task scope.",
        "model": model, "effort": effort, "settings": reference(scoped_settings),
        "limits": "This changes model selection. It is the same retained case, not an independent fresh qualification. Observe actual parent and child model IDs, completed child delivery and caller result. No provider/art/human acceptance.",
        "human_acceptance": "separate_unperformed", "open_milestones": ["M1", "M2", "M3", "M4", "M5"],
        "preparer": reference(Path(__file__)),
        "source": {str(p.relative_to(ROOT)): reference(p) for p in
                   (ROOT / "evaluations/native_trial_records.py", ROOT / "evaluations/request_review_trials.py",
                    ROOT / "evaluations/cold_caller_review.py")},
    })
    handoff._copy(previous / "project", destination / "input-project", current)
    archive = handoff._archive(destination / "input-project", destination / "input-project.zip", current)
    identity_path = destination / "identity.json"
    identity = json.loads(identity_path.read_bytes())
    identity.update(case=old_identity["case"], consumer=str(consumer), inputs=current,
        input_archive=archive, static_inputs=old_identity["static_inputs"],
        package=old_identity["package"], package_files=old_identity["package_files"],
        source_revision=old_identity["source_revision"],
        registered_agent=old_identity["registered_agent"], target_role=old_identity["target_role"],
        original_inputs=old_identity["original_inputs"],
        predecessor_identity=reference(previous / "identity.json"),
        predecessor_preservation=reference(previous / "preservation.json"),
        predecessor_files=predecessor_files, continuation_protocol=reference(protocol_path),
        retention_scope="Same native case; exact 167-file entry snapshot, unchanged 261-file kit; selected host model explicitly restored. Generic recorder execution followed by full-project retention.")
    write_json(identity_path, identity)
    assert inventory(previous, ("__pycache__",)) == predecessor_files
    assert current == handoff._files(consumer, exclude_kit=True)
    assert reference(user_settings_path) == settings_before
    print(json.dumps({"prepared": str(destination), "files": len(current), "model": model,
                      "auth": selected_auth, "launched": False}))


if __name__ == "__main__":
    main()
