def sha(data):
    return hashlib.sha256(data).hexdigest()


def ref(path, lines=None, pointer=None):
    data = path.read_bytes()
    value = {"path": path.relative_to(ROOT).as_posix(), "sha256": sha(data), "bytes": len(data)}
    if lines:
        assert 1 <= lines[0] <= lines[1] <= len(data.decode("utf-8").splitlines())
        value["lines"] = lines
    if pointer:
        value["json_pointer"] = pointer
    return value


def read(path):
    return json.loads(path.read_bytes())


def finals(path):
    rows = [json.loads(line) for line in path.read_text(encoding="utf-8").splitlines()]
    result = []
    for number, row in enumerate(rows, 1):
        msg = row.get("message", {})
        if row.get("type") == "assistant" and msg.get("stop_reason") == "end_turn":
            text = "\n".join(item["text"] for item in msg.get("content", []) if item.get("type") == "text")
            if text:
                result.append({"line": number, "text": text, "text_sha256": sha(text.encode())})
    return rows, result
