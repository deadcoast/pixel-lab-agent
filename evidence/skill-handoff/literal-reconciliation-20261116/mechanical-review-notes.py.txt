"""Retain final attribution and installed-cache observations without changing native evidence."""
from datetime import UTC, datetime
import hashlib
import json
from pathlib import Path
import zipfile

ROOT = Path(__file__).resolve().parents[3]
BASE = Path(__file__).parent
ATTEMPT = BASE / 'native-01'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


def rows(path):
    return [json.loads(line) for line in path.read_text(encoding='utf-8').splitlines()]


def ref(path):
    return {'path': path.relative_to(ROOT).as_posix(), 'sha256': sha(path)}


def write(path, value):
    with path.open('x', encoding='utf-8') as stream:
        json.dump(value, stream, indent=2)
        stream.write('\n')


def inventory(root):
    return {path.relative_to(root).as_posix(): {'sha256': sha(path),
        'bytes': path.stat().st_size, 'mtime_ns': path.stat().st_mtime_ns}
        for path in sorted(root.rglob('*')) if path.is_file()}


def blocks(row):
    content = row.get('message', {}).get('content', [])
    return content if isinstance(content, list) else []


identity = read(ATTEMPT / 'identity.json')
consumer = Path(identity['consumer'])
kit = consumer / 'pixl-kit'

assert mech['checker']['sha256'] == 'dfa7ee588561365055d4bcd2d4c555865dbe9f603b40cbeb296797a85b41b116'

parent, child, stream = rows(parent_path), rows(child_path), rows(ATTEMPT / 'stdout.jsonl')
before = inventory(kit)
payload = identity['package_files']
added = sorted(set(before) - set(payload))
missing = sorted(set(payload) - set(before))
changed = sorted(name for name in set(payload) & set(before) if before[name]['sha256'] != payload[name])
assert len(payload) == 257 and len(added) == 17 and not missing and not changed
assert all('__pycache__' in Path(name).parts and name.endswith('.pyc') for name in added)
cache_archive = BASE / 'runtime-observation-cache-01.zip'
with zipfile.ZipFile(cache_archive, 'x', zipfile.ZIP_DEFLATED) as archive:
    for name in added:
        archive.writestr(name, (kit / name).read_bytes())
with zipfile.ZipFile(cache_archive) as archive:
    assert len(archive.namelist()) == len(set(archive.namelist())) == len(added)
    assert set(archive.namelist()) == set(added)
    assert all(hashlib.sha256(archive.read(name)).hexdigest() == before[name]['sha256'] for name in added)
after = inventory(kit)
assert before == after
direct_calls = {}
session_writes = {}
for number, row in enumerate(child, 1):
    for block in blocks(row):
        if block.get('type') != 'tool_use':
            continue
        inp = block.get('input', {})
        command = inp.get('command', '')
        if ' -I pixl-kit/scripts/pixl.py ' in command:
            direct_calls.setdefault(block['id'], {'first_native_line': number, 'command': command})
        if block.get('name') == 'Write' and inp.get('file_path', '').endswith('/SESSION.md'):
            text = inp['content']
            session_writes.setdefault(block['id'], {'first_native_line': number,
                'utf8_payload_sha256': hashlib.sha256(text.encode()).hexdigest(), 'content': text})
