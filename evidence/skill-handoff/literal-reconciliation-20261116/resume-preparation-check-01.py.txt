from datetime import UTC, datetime
import hashlib
import json
from pathlib import Path
import zipfile

ROOT = Path(__file__).resolve().parents[3]

PREVIOUS = BASE / 'native-02'
ATTEMPT = BASE / 'native-02-resume-01'


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def read(path):
    return json.loads(path.read_bytes())


def ref(path):
    return {'path': str(path.resolve()), 'sha256': sha(path), 'bytes': path.stat().st_size}


def files(root, exclude=()):
    return {p.relative_to(root).as_posix(): sha(p) for p in root.rglob('*')
            if p.is_file() and not any(part in exclude for part in p.relative_to(root).parts)}


def archive(path):
    with zipfile.ZipFile(path) as source:
        names = source.namelist()
        assert len(names) == len(set(names)), 'Duplicate archive entries'
        return {name: hashlib.sha256(source.read(name)).hexdigest() for name in names if not name.endswith('/')}
