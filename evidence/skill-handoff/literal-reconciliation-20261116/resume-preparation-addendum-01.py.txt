    return {'path': str(path.resolve()), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest(), 'bytes': path.stat().st_size}


preparation = json.loads((BASE / 'resume-preparation-01.json').read_bytes())
original = ref(BASE / 'mechanical-03.json')
assert original['sha256'] == '21db78d788762ab408bd4b63d0f3b152ef5aa7b277d36a0a16fa6c3ecee387bb'
checker = ref(BASE / 'mechanical-check-04.py.txt')
assert checker == preparation['mechanical_checker']
source = (BASE / 'mechanical-check-04.py.txt').read_text(encoding='utf-8')
keys = ['same_native_parent_resumed_without_fork', 'same_actual_child_continued_without_replacement',
        'resumed_parent_retains_original_journal_prefix', 'resumed_child_retains_original_journal_prefix',
        'corrected_child_return_after_resume_delivered']
assert all('check("' + key + '"' in source for key in keys)
assert not (BASE / 'native-02-resume-01' / 'stdout.jsonl').exists()
payload = {
    'observed_at': datetime.now(UTC).isoformat(), 'status': 'pass',
    'preparation': ref(BASE / 'resume-preparation-01.json'), 'mechanical_checker': checker,
    'additional_prepared_bindings': {'original_incomplete_mechanical_receipt': original},
    'required_continuity_and_return_check_keys': keys,

This is retained-file association only, never proof that the failed invocation produced the file. Final independent review must bind actual native delivered output, successful retry and saved report separately, preserve failed outputs and identify any reused report path. Structural coverage never certifies source meaning.',

    'recorder': ref(Path(__file__)),
}
with (BASE / 'resume-preparation-addendum-01.json').open('x', encoding='utf-8', newline='\n') as destination:
    json.dump(payload, destination, indent=2, ensure_ascii=False)
