"""Freeze independent static review of the added handoff coverage operation."""
import argparse
from datetime import UTC, datetime
import hashlib
import json
from pathlib import Path
import subprocess

ROOT = Path(__file__).resolve().parents[3]
BASE = Path(__file__).parent


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path):
    return {'path': path.relative_to(ROOT).as_posix(), 'sha256': sha(path)}


parser = argparse.ArgumentParser()
parser.add_argument('--handoff-guide-sha256', required=True)
args = parser.parse_args()
paths = ['pixl/handoff_review.py', 'plugins/pixl/scripts/check_handoff.py',
    'plugins/pixl/skills/pixl/references/delegation.md',
    'plugins/pixl/skills/pixl/references/intake-sample.md',
    'plugins/pixl/skills/pixl/references/handoff-review.md', 'tests/test_handoff_review.py']
hashes = {name: sha(ROOT / name) for name in paths}
assert hashes[paths[0]] == '5f49e8f3514bc59117714f7392528b9f60f1aa0ba874d5caef262a093153c7d4'
assert hashes[paths[1]] == 'ac1afd38549c6e951ab02afdfcc1695f3dcd309b3e357bca524a679271c02acc'
assert hashes[paths[2]] == '72bd2cdfd03333ae0cc3bcd325c4a231f0730ca2f42fbece1955b18b5648199b'
assert hashes[paths[3]] == '9fd421030d44e104ff36446409d4562a976daec1650f7a956a950ae8043846ef'
assert hashes[paths[4]] == args.handoff_guide_sha256
old_review = json.loads((BASE / 'source-review-01.json').read_bytes())
for item in old_review['reviewed_sources']:
    assert sha(ROOT / item['path']) == item['sha256']
baseline = json.loads((BASE / 'baseline.json').read_bytes())['head']
tracked = subprocess.check_output(['git', 'ls-tree', '-r', '--name-only', baseline, 'pixl'], cwd=ROOT, text=True).splitlines()
existing_changes = [name for name in tracked if (ROOT / name).read_bytes() != subprocess.check_output(['git', 'show', baseline + ':' + name], cwd=ROOT)]
actual_runtime = {p.relative_to(ROOT).as_posix() for p in (ROOT / 'pixl').rglob('*') if p.is_file() and '__pycache__' not in p.parts}
assert not existing_changes and actual_runtime - set(tracked) == {'pixl/handoff_review.py'}

result = {'observed_at': datetime.now(UTC).isoformat(), 'status': 'pass',
    'review_type': 'independent manual code/source and guidance review', 'reviewer': '/root/prerequisite_mechanics',
    'source_sha256': hashes, 'baseline': baseline, 'existing_runtime_source_changes': existing_changes,
