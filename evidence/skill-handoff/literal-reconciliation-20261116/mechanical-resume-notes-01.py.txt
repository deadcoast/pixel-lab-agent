ATTEMPT = BASE / 'native-02-resume-01'
PREVIOUS = BASE / 'native-02'


def read(path):
    return json.loads(path.read_bytes())


def ref(path):
    return {'path': str(path.resolve()), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest(), 'bytes': path.stat().st_size}


def rows(path):
    return [json.loads(line) for line in path.read_text(encoding='utf-8').splitlines()]


def texts(row):
    content = row.get('message', {}).get('content', [])
    return [content] if isinstance(content, str) else [part['text'] for part in content if part.get('type') == 'text']


def error_rows(source, start=0):
    return [{'line': index, 'error': row.get('error'), 'model': row.get('message', {}).get('model'),
             'stop_reason': row.get('message', {}).get('stop_reason'), 'text': texts(row)}
            for index, row in enumerate(source, 1) if index > start and row.get('error')]
