def read(name):
    raw = (ROOT / name).read_bytes()
    return raw.decode().replace("\r\n", "\n"), "\r\n" if b"\r\n" in raw else "\n"


def write(name, text, newline):
    (ROOT / name).write_bytes(text.replace("\n", newline).encode())


def replace_line(text, prefix, replacement):
    lines = text.splitlines()
    matches = [i for i, line in enumerate(lines) if line.startswith(prefix)]
    if len(matches) != 1:
        raise ValueError(f"Expected one {prefix!r}; got {len(matches)}")
    lines[matches[0]] = replacement
    return "\n".join(lines) + "\n"
