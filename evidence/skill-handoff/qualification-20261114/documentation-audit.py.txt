"""Check current navigation, retained results and the existing milestone cycle."""
import argparse
from datetime import UTC, datetime
import hashlib
import json
from pathlib import Path
import re
import subprocess
from urllib.parse import unquote

ROOT = Path(__file__).resolve().parents[3]
BASE = "f9103cd532ce03cf4eb61520c02ad9d68f1d40a2"
DOCS = ["TODO.md", "PLAN.md", "SHELVED.md", "docs/IDENTIFIED.md",
        "docs/CAPABILITY-OUTCOMES.md", "evaluations/TODO.spec.md",
        "evidence/README.md", "evidence/skill-handoff/README.md",
        "evidence/skill-handoff/qualification-20261114/README.md"]


def sha(data):
    return hashlib.sha256(data).hexdigest()


def original(name):
    return subprocess.check_output(["git", "show", BASE + ":" + name], cwd=ROOT)


def anchors(path):
    body = path.read_text(encoding="utf-8-sig")
    result = set(re.findall(r'<a\s+id=["\']([^"\']+)["\']', body))
    counts = {}
    for heading in re.findall(r"^#{1,6}\s+(.+?)\s*#*$", body, re.M):
        slug = re.sub(r"[^\w\s-]", "", heading.lower()).replace(" ", "-")
        count = counts.get(slug, 0)
        counts[slug] = count + 1
        result.add(slug + (f"-{count}" if count else ""))
    return result


def main():
    parser = argparse.ArgumentParser()
    parser.add_argument("--out", type=Path, required=True)
    args = parser.parse_args()
    checks = []

    def check(name, passed, detail=None):
        checks.append({"name": name, "passed": bool(passed), "detail": detail})

    for name in DOCS:
        path = ROOT / name
        body = re.sub(r"(?ms)^(`{3,}|~{3,})[^\n]*\n.*?^\1[ \t]*(?:\n|$)", "", path.read_text(encoding="utf-8-sig"))
        for target in re.findall(r"\]\(([^)]+)\)", body):
            if re.match(r"[a-zA-Z][a-zA-Z0-9+.-]*:", target):
                continue
            local, _, fragment = unquote(target).partition("#")
            destination = (path.parent / local).resolve() if local else path
            exists = destination.exists() or destination == args.out.resolve()
            valid = not fragment or (destination.is_file() and fragment in anchors(destination))
            check(f"link:{name}:{target}", exists and valid)
    pattern = r"^- \*\*((?:NF|SC|QOL|CF|TM)-\d+)\s"
    prior = re.findall(pattern, original("docs/IDENTIFIED.md").decode(), re.M)
    current = re.findall(pattern, (ROOT / "docs/IDENTIFIED.md").read_text(encoding="utf-8"), re.M)
    backlog = re.findall(r"^\| \[((?:NF|SC|QOL|CF|TM)-\d+)\]", (ROOT / "SHELVED.md").read_text(encoding="utf-8"), re.M)
    check("fifty_original_ids", len(current) == len(set(current)) == len(backlog) == len(set(backlog)) == 50 and set(prior) == set(current) == set(backlog))
    for name in DOCS[:4]:
        body = (ROOT / name).read_text(encoding="utf-8")
        check("active_cycle:" + name, "IDENTIFY-2026-11-09-01" in body and "PX-SKILL-001" in body and "PX-EVAL-005" in body)
    plan = (ROOT / "PLAN.md").read_text(encoding="utf-8")
    sections = dict(re.findall(r"(?ms)^## (M[1-5]) [^\n]+\n(.*?)(?=^## |\Z)", plan))
    for milestone in ("M1", "M2", "M3", "M4", "M5"):
        status = re.search(r"^\*\*Status:\*\* ([^\n]+)", sections.get(milestone, ""), re.M)
        body = status.group(1).split("**Existing work:")[0] if status else ""
        check("milestone_open:" + milestone, bool(re.search(r"\b(open|active)\b", body, re.I)) and not re.search(r"\b(closed|complete)\b", body, re.I), body)
    before = json.loads(original("evaluations/evidence-claims.json"))
    after = json.loads((ROOT / "evaluations/evidence-claims.json").read_bytes())
    for kind in ("claims", "receipts"):
        current = {row["id"]: row for row in after[kind]}
        check("old_registry_objects_unchanged:" + kind, all(current.get(row["id"]) == row for row in before[kind]))
    check("release_declarations_unchanged", before["release"] == after["release"])
    for claim in before["claims"]:
        doc = claim["document"]
        pinned = [line for line in (ROOT / doc["path"]).read_text(encoding="utf-8").splitlines() if f'id="{doc["anchor"]}"' in line]
        check("pinned_claim_line:" + claim["id"], len(pinned) == 1 and sha(pinned[0].encode()) == doc["line_sha256"])
    baseline = json.loads((Path(__file__).parent / "baseline.json").read_bytes())
    for name, digest in baseline["prior_handoff_files"].items():
        if name != "evidence/skill-handoff/README.md":
            check("retained_handoff:" + name, sha((ROOT / name).read_bytes()) == digest)
    revision = json.loads((Path(__file__).parent / "revision-05/RESULT.json").read_bytes())
    changed_production = {"plugins/pixl/" + name for name in revision["changed_package_paths"]}
    check("only_declared_guidance_and_manifest_changed", changed_production == {
        "plugins/pixl/skills/pixl/SKILL.md", "plugins/pixl/skills/pixl/references/intake-sample.md",
        "plugins/pixl/bundle-manifest.json"} and not revision["runtime_changed"])
    for name, digest in baseline["source_files"].items():
        if name in changed_production:
            digest = revision["package_files"][name.removeprefix("plugins/pixl/")]
        check("tested_production_source:" + name, sha((ROOT / name).read_bytes()) == digest)
    changed = subprocess.check_output(["git", "diff", "--name-only", "--diff-filter=MDTR", BASE], text=True, cwd=ROOT).splitlines()
    allowed_evidence = {"evidence/README.md", "evidence/skill-handoff/README.md"}
    check("other_historical_evidence_unchanged", not [p for p in changed if p.startswith("evidence/") and p not in allowed_evidence])
    result = {"observed_at": datetime.now(UTC).isoformat(), "baseline": BASE,
              "scope": "Documentation links, original finding IDs, open milestones, unchanged historical receipts, revised tested guidance and otherwise unchanged production sources. No replay or human acceptance.",
              "script_sha256": sha(Path(__file__).read_bytes()),
              "documents": {name: sha((ROOT / name).read_bytes()) for name in DOCS},
              "status": "pass" if all(c["passed"] for c in checks) else "fail",
              "passed": sum(c["passed"] for c in checks), "total": len(checks), "checks": checks}
    with args.out.open("x", encoding="utf-8") as stream:
        json.dump(result, stream, indent=2)
        stream.write("\n")
    print(json.dumps({key: result[key] for key in ("status", "passed", "total")}))
    if result["status"] != "pass":
        print(json.dumps([c for c in checks if not c["passed"]]))
        raise SystemExit(1)


if __name__ == "__main__":
    main()
