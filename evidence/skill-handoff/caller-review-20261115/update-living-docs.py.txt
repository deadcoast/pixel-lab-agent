def change(name, transform):
    path = ROOT / name
    raw = path.read_bytes()
    newline = "\r\n" if b"\r\n" in raw else "\n"
    body = raw.decode("utf-8").replace("\r\n", "\n")
    revised = transform(body)
    if revised == body:
        raise ValueError("Expected a contextual update: " + name)
    path.write_bytes(revised.replace("\n", newline).encode("utf-8"))


def line(body, prefix, replacement):
    rows = body.splitlines()
    matches = [index for index, value in enumerate(rows) if value.startswith(prefix)]
    if len(matches) != 1:
        raise ValueError("Expected one line starting " + prefix)
    rows[matches[0]] = replacement
    return "\n".join(rows) + "\n"
