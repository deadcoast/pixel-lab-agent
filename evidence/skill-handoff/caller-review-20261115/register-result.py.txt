"""Append the autonomous caller-review outcome without changing prior claims."""
import hashlib
import json
from pathlib import Path

ROOT = Path(__file__).resolve().parents[3]
PREFIX = "evidence/skill-handoff/caller-review-20261115/"


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def main():
    path = ROOT / "evaluations/evidence-claims.json"
    registry = json.loads(path.read_bytes())
    identifier = "PX-SKILL-001-NATIVE-CALLER-REVIEW"
    if any(claim["id"] == identifier for claim in registry["claims"]):
        raise ValueError("Caller-review outcome already registered")
    entries = {
        "pxskill001-caller-result": "RESULT.json",
        "pxskill001-caller-semantic-initial": "semantic-review-01.json",
        "pxskill001-caller-semantic-revised": "semantic-review-02.json",
        "pxskill001-caller-mechanical-initial": "mechanical-02.json",
        "pxskill001-caller-mechanical-revised": "mechanical-04.json",
        "pxskill001-caller-local": "local-02/RESULT.json",
        "pxskill001-caller-seed": "seed-preflight/RESULT.json",
        "pxskill001-caller-preparation-initial": "preparation-01.json",
        "pxskill001-caller-preparation-revised": "preparation-02.json",
        "pxskill001-caller-package": "package-check-01/RESULT.json",
        "pxskill001-caller-revision": "revision-01/RESULT.json",
    }
    for receipt_id, name in entries.items():
        file = ROOT / PREFIX / name
        value = json.loads(file.read_bytes())
        registry["receipts"].append({"id": receipt_id, "path": PREFIX + name, "sha256": sha(file),
            "assertions": [{"pointer": "/status", "equals": value["status"]}]})
    document, anchor = "evidence/skill-handoff/README.md", "pxskill001-native-caller-review"
    lines = (ROOT / document).read_text(encoding="utf-8").splitlines()
    pinned = [line for line in lines if f'id="{anchor}"' in line]
    if len(pinned) != 1:
        raise ValueError("Expected one exact dated caller-review claim")
    result = json.loads((ROOT / PREFIX / "RESULT.json").read_bytes())
    registry["claims"].append({
        "id": identifier, "task": "Native caller/specialist cold review without maintainer feedback",
        "document": {"path": document, "anchor": anchor,
                     "line_sha256": hashlib.sha256(pinned[0].encode()).hexdigest()},
        "tier": "native registered caller/specialist workflow, independent semantic and artifact review",
        "scope": result["scope"],
        "limits": ["Two recorded Claude workflows from the exact retained cold04 revision6; no general host-reliability claim.",
                   "The actual criterion outcomes and any unexercised branches remain explicit in the semantic receipt.",
                   "Prior unaided direct-profile failures and maintainer-guided correction retain their original outcomes.",
                   "Local source/artifact checks do not establish provider generation, engine acceptance or artistic approval.",
                   "Human acceptance is separate and unperformed; M1/M2 and broader M3/M4/M5 remain open."],
        "policy": "current", "outcome": {"receipt": "pxskill001-caller-result", "pointer": "/status"},
        "receipts": list(entries),
        "bindings": [{"kind": "map", "domain": "source", "receipt": "pxskill001-caller-result",
                      "pointer": "/source_hashes", "prefix": ""},
                     {"kind": "map", "domain": "source", "receipt": "pxskill001-caller-result",
                      "pointer": "/protocol_hashes", "prefix": ""},
                     {"kind": "map", "domain": "source", "receipt": "pxskill001-caller-result",
                      "pointer": "/evidence_hashes", "prefix": ""},
                     {"kind": "file", "domain": "package", "path": "plugins/pixl/bundle-manifest.json",
                      "receipt": "pxskill001-caller-result", "pointer": "/bundle_manifest_sha256"}]})
    path.write_text(json.dumps(registry, indent=2, ensure_ascii=False) + "\n", encoding="utf-8", newline="\n")
    print(json.dumps({"claims": len(registry["claims"]), "receipts": len(registry["receipts"]),
                      "registered_outcome": result["status"]}))


if __name__ == "__main__":
    main()
