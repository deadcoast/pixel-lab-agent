def digest(data):
    return hashlib.sha256(data).hexdigest()


def sha(path):
    return digest(path.read_bytes())


def read(path):
    return json.loads(path.read_bytes())


def reference(path):
    return {"path": path.resolve().relative_to(ROOT).as_posix(), "sha256": sha(path),
            "bytes": path.stat().st_size}


def rows(path):
    return [dict(json.loads(line), file=str(path.resolve()), line=index,
                 line_sha256_without_newline=digest(line))
            for index, line in enumerate(path.read_bytes().splitlines(), 1) if line.strip()]


def blocks(row):
    value = row.get("message", {}).get("content", [])
    return value if isinstance(value, list) else []


def one(values, label):
    values = list(values)
    if len(values) != 1:
        raise ValueError(f"Expected one {label}, observed {len(values)}")
    return values[0]


def tag(text, name):
    return one(re.findall("<" + name + ">(.*?)</" + name + ">", text, re.S), name)


def event(row):
    return {"path": Path(row["file"]).relative_to(ROOT).as_posix(), "line": row["line"],
            "line_sha256_without_newline": row["line_sha256_without_newline"],
            **{k: row[k] for k in ("type", "subtype", "session_id", "sessionId", "agentId",
                "task_id", "run_id", "tool_use_id", "status", "origin") if k in row}}


def confined(root, name):
    path = (root / name).resolve()
    if not path.is_relative_to(root):
        raise ValueError("Consumer reference escaped root")
    return path


def ref_matches(root, value, expected=None):
    path = confined(root, value["path"])
    return sha(path) == value["sha256"] and (expected is None or path == expected)


def first(mapping, *keys):
    for key in keys:
        if key in mapping:
            return mapping[key]
    raise KeyError("Missing semantic field: " + "/".join(keys))


def sequence(value):
    return [value[key] for key in EDGES] if isinstance(value, dict) else value


def png_metrics(root, names):
    frames = []
    for name in names:
        with Image.open(root / name) as image:
            if image.size != (32, 32):
                raise ValueError("Unexpected actual frame dimensions")
            frames.append([tuple(pixel) if pixel[3] else (0, 0, 0, 0)
                           for pixel in image.convert("RGBA").getdata()])
    alpha = [[pixel[3] > 0 for pixel in frame] for frame in frames]
    delta = lambda a, b: sum(x != y for x, y in zip(a, b, strict=True))
    return {"frames": names,
            "baseline_vs_frame0_alpha_xor": [delta(alpha[0], a) for a in alpha],
            "baseline_vs_frame0_rgba_changed": [delta(frames[0], f) for f in frames],
            "adjacent_alpha_xor": {edge: delta(alpha[i], alpha[(i + 1) % 5]) for i, edge in enumerate(EDGES)},
            "adjacent_rgba_changed": {edge: delta(frames[i], frames[(i + 1) % 5]) for i, edge in enumerate(EDGES)}}

if __name__ == "__main__":
