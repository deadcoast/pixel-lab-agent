def read(path):
    return json.loads(path.read_bytes())


def sha(path):
    return hashlib.sha256(path.read_bytes()).hexdigest()


def ref(path):
    return {"path": path.resolve().relative_to(ROOT).as_posix(), "sha256": sha(path), "bytes": path.stat().st_size}


def blocks(row):
    content = row.get("message", {}).get("content", [])
    return content if isinstance(content, list) else []

    if out.exists():

                uses[block["id"]] = (row, block)
                item = {"line": row["line"], "actor": row.get("agent_id", "parent"), "tool": tool}
                if tool == "Bash":
                    digest = hashlib.sha256(args["command"].encode()).hexdigest()
                    item.update(command_sha256=digest, review=COMMANDS.get(digest), allowed=digest in COMMANDS)

                        "mcp__pixl-unknown__pixl_jobs", "mcp__pixl-unknown__pixl_ready",
                        "mcp__pixl-unknown__pixl_job_dispatch"}

                if use and use[1]["name"] == "Bash":

    refs = []
    def visit(value):
        if isinstance(value, dict):

                target = (root / value["path"]).resolve()
                refs.append({**value, "matches": target.is_relative_to(root) and target.is_file() and sha(target) == value["sha256"]})

                visit(v)
        elif isinstance(value, list):

    }
