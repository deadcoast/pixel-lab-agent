"""Independent read-only checks of one frozen native authoring attempt."""
import base64
import json
import sys
from hashlib import sha256
from pathlib import Path

repo = Path(__file__).resolve().parents[1]
host = sys.argv[1]
attempt = repo / 'evidence/native-authoring/20261109/native' / (host + '-01')
root = attempt / 'project'
kit = root / '.agents/skills/pixl-kit'
sys.path.insert(0, str(kit / 'runtime'))
from pixl import codec, jobs, plar, production_plan, store


def read(path):
    return json.loads(path.read_text(encoding='utf-8'))


def digest(path):
    return sha256(path.read_bytes()).hexdigest()


def changed(parent, inventory):
    return [name for name, expected in inventory.items()
            if not (parent / name).is_file() or digest(parent / name) != expected]


identity = read(attempt / 'identity.json')
origin = read(attempt / 'origin.json')
execution = read(attempt / 'record.json')
caller = read(root / 'AUTHORING-RESULT.json')
folder = root / '.pixl/assets/queue.ready'
doc = store.load(folder)
old = store.load(attempt / 'input-project/.pixl/assets/queue.ready')
old_changed = [ident for ident, record in old.records.items()
               if ident not in doc.records or codec.record_hash(record) != codec.record_hash(doc.records[ident])]
new = {ident: record for ident, record in doc.records.items() if ident not in old.records}
markers = [r for r in new.values() if r.fields.get('mt', {}).get('binding') == 'pixl.production.plan/1']
assert len(markers) == 1, 'Expected exactly one new connected plan'
marker = markers[0]
snapshot = read(root / marker.fields['ur'])
plan = snapshot['plan']
plan_id = plan['id']
stages = plan['stages']
stage_by_asset = {s['selection']['asset']: s for s in stages if 'selection' in s}
working = stage_by_asset['queue.working']
working_task = plan_id + '.' + working['id'] + '.task'
report = jobs.report(doc, root, folder)
new_jobs = [j for j in report['jobs'] if j['id'] in new]
assert len(new_jobs) == 1, 'Expected exactly one new durable job'
job = new_jobs[0]
request = job['request']
packet_records = [r for r in new.values() if r.kind == 'X' and '/stage-requests/' in r.fields.get('ur', '')]
assert len(packet_records) == 1
packet_record = packet_records[0]
packet_saved = read(root / packet_record.fields['ur'])
packet = packet_saved['packet'] if 'packet' in packet_saved else packet_saved
events = [json.loads(s) for s in (attempt / 'evaluation-events.jsonl').read_text().splitlines()]
allocation = read(root / 'authoring-plar.txt')
selected = {a['id']: a for a in allocation['assets']}
selection_status = {asset: plar.read(root, 'authoring-plar.txt', asset, include_production=False)
                    for asset in stage_by_asset}
links = {}
for asset, stage in stage_by_asset.items():
    links[asset] = selected[asset].get('production')

input_candidate = doc.records['live.working-seed.candidate']
input_hash = digest(root / input_candidate.fields['ur'])
arguments = request['arguments']
embedded = arguments.get('first_frame_base64')
argv = arguments.get('argv', [])
subjects = [x.split(',', 1)[1] for x in argv if x.startswith('data:image/png;base64,')] if isinstance(argv,list) else []
embedded_hashes = [sha256(base64.b64decode(x, validate=True)).hexdigest() for x in ([embedded] if embedded else subjects)]
animation = request['tool'].split('__')[-1] == 'animate_image' and arguments.get('frame_count') == 4 and not arguments.get('first_frame_url') and bool(embedded)
join = request['tool'].split('__')[-1] == 'pixelart_workbench' and isinstance(argv,list) and argv[:1] == ['join'] and len(subjects) == 4 and len(argv) == 9 and argv[5:] == ['--as','sheet','--columns','4']
discovery = request['discovery']['evidence']
origin_changed = changed(root, origin['files'])
kit_changed = changed(kit, identity['package_files'])
all_input_changed = changed(root, identity['inputs'])
new_tasks = {ident: r.fields['st'].value for ident, r in new.items() if r.kind == 'T'}
pointers = []


def walk(value):
    if isinstance(value, dict):
        if isinstance(value.get('path'), str) and isinstance(value.get('sha256'), str):
            p = root / value['path']
            pointers.append({'path': value['path'], 'matches': p.is_file() and digest(p) == value['sha256']})
        for child in value.values():
            walk(child)
    elif isinstance(value, list):
        for child in value:
            walk(child)


walk(caller)
checks = {
    'native_process_exit_zero': execution['returncode'] == 0,
    'original_files_changed_only_append_only_state': origin_changed == ['.pixl/assets/queue.ready/state.px'],
    'all_original_state_records_unchanged': not old_changed,
    'seeded_inputs_changed_only_state_and_planning_branch': set(all_input_changed) == {'.pixl/assets/queue.ready/state.px', 'authoring-plar.txt'},
    'frozen_kit_unchanged': not kit_changed,
    'one_plan_and_one_job': len(markers) == len(new_jobs) == 1,
    'only_three_requested_asset_stages': set(stage_by_asset) == {'queue.working','queue.retry','queue.paused'},
    'all_new_tasks_open_without_verdicts': all(v == 'open' for v in new_tasks.values()) and not any(r.kind == 'V' for r in new.values()),
    'working_exact_seed_prerequisite': working['needs'] == ['live.seed.output'],
    'packet_exact_seed_identity': packet['inputs'] == {'live.seed.output': {'candidate': 'live.working-seed.candidate', 'sha256': input_hash}},
    'request_embeds_exact_seed_bytes': bool(embedded_hashes) and all(h == input_hash for h in embedded_hashes),
    'supported_declared_operation_and_cardinality': animation or join,
    'frozen_discovery_identity': discovery.get('sha256',discovery.get('declaration_sha256')) == digest(root/'official-discovery.json'),
    'exact_plan_snapshot_hash': digest(root/marker.fields['ur']) == marker.fields['dg'],
    'exact_stage_packet_hash': digest(root/packet_record.fields['ur']) == packet_record.fields['dg'],
    'three_recovered_allocation_links': all(link and link['task'] == plan_id+'.'+stage_by_asset[asset]['id']+'.task' for asset,link in links.items()),
    'both_controlled_events_observed_once': [e['kind'] for e in events] == ['allocation_write_interrupted','caller_requirement_updated'],
    'working_requirement_changed_and_others_current': all((selection_status[a]['requirements_hash'] == s['selection']['requirements_hash']) == (a != 'queue.working') for a,s in stage_by_asset.items()),
    'request_retained_prepared_without_remote_id': job['state'] == 'prepared' and not job['remote_id'],
    'old_working_request_blocked_by_current_requirements': bool(job['blockers']) and job['next_action'] == 'resolve_context' and bool(production_plan.blockers(doc,working_task,root,folder)),
    'all_caller_path_hash_pointers_resolve': bool(pointers) and all(p['matches'] for p in pointers),
}
out = repo / 'evidence/native-authoring/independent-review'
out.mkdir(exist_ok=True)
result = {
    'host': host, 'status': 'pass' if all(checks.values()) else 'fail', 'checks': checks,
    'scope': 'Read-only identity/state checks using the frozen copied runtime. Behavior, semantics and caller independence are separately reviewed.',
    'runtime_source': str(kit.relative_to(repo)), 'state_revision': doc.revision, 'state_sha256': codec.digest(doc),
    'original_record_count': len(old.records), 'new_record_count': len(new),
    'changed_original_records': old_changed, 'changed_origin': origin_changed,
    'changed_inputs': all_input_changed, 'changed_kit': kit_changed,
    'plan_id': plan_id, 'plan_snapshot': marker.fields['ur'], 'stage_packet': packet_record.fields['ur'],
    'job': job['id'], 'job_state': job['state'], 'blockers': job['blockers'], 'links': links,
    'original_working_requirements': working['selection']['requirements_hash'],
    'current_working_requirements': selection_status['queue.working']['requirements_hash'],
    'input_sha256': input_hash, 'embedded_input_sha256': embedded_hashes,
    'caller_result_sha256': digest(root/'AUTHORING-RESULT.json'), 'caller_pointers': pointers,
    'events': events,
}
(out/(host+'-checks.json')).write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8')
print(json.dumps({'host':host,'status':result['status'],'checks':checks},indent=2))
